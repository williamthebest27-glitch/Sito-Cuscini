if (!class_exists('TDTNav_Desktop_Walker') && class_exists('Walker_Nav_Menu')) {
	class TDTNav_Desktop_Walker extends Walker_Nav_Menu {
		public function start_lvl(&$output, $depth = 0, $args = null) {
			$output .= '<ul class="dtnav__sub">';
		}
		public function end_lvl(&$output, $depth = 0, $args = null) {
			$output .= '</ul>';
		}
		public function start_el(&$output, $item, $depth = 0, $args = null, $id = 0) {
			$has = in_array('menu-item-has-children', (array) $item->classes, true);
			if ($depth === 0) {
				$output .= '<li class="dtnav__item' . ($has ? ' dtnav__item--has' : '') . '">';
				$output .= '<a href="' . esc_url($item->url) . '" class="dtnav__link">' . esc_html($item->title);
				if ($has) { $output .= '<span class="dtnav__caret" aria-hidden="true">&#9662;</span>'; }
				$output .= '</a>';
			} else {
				$output .= '<li><a href="' . esc_url($item->url) . '" class="dtnav__sublink">' . esc_html($item->title) . '</a>';
			}
		}
		public function end_el(&$output, $item, $depth = 0, $args = null) {
			$output .= '</li>';
		}
	}
}

/* -----------------------------------------------------------
 * 3) Walker MOBILE — voci numerate, tutte sullo stesso piano
 *
 *    Da telefono i sottomenu non ci sono: le voci figlie
 *    ("Scopri i prodotti", "Carrello", "Pagamento") salgono in
 *    lista come tutte le altre e la voce che le teneva insieme
 *    ("Negozio") non viene stampata. Una lista sola, un tocco solo
 *    per arrivare dove si vuole. Da monitor la tendina resta.
 * --------------------------------------------------------- */
if (!class_exists('TDTNav_Mobile_Walker') && class_exists('Walker_Nav_Menu')) {
	class TDTNav_Mobile_Walker extends Walker_Nav_Menu {
		private $conta = 0;
		/* niente lista annidata: i figli proseguono in quella di sopra */
		public function start_lvl(&$output, $depth = 0, $args = null) {}
		public function end_lvl(&$output, $depth = 0, $args = null) {}
		private function cartella($item, $depth) {
			return $depth === 0 && in_array('menu-item-has-children', (array) $item->classes, true);
		}
		public function start_el(&$output, $item, $depth = 0, $args = null, $id = 0) {
			/* la voce che fa da cartella non si stampa: al suo posto
			   compaiono i suoi figli, numerati come le altre */
			if ($this->cartella($item, $depth)) { return; }
			$this->conta++;
			$indice = str_pad($this->conta, 2, '0', STR_PAD_LEFT);
			$output .= '<li class="dtnav-menu__item">';
			$output .= '<a href="' . esc_url($item->url) . '" class="dtnav-menu__link" data-dtnav-close>'
			         . '<span class="dtnav-menu__index">' . esc_html($indice) . '</span>'
			         . '<span class="dtnav-menu__label">' . esc_html($item->title) . '</span>'
			         . '<span class="dtnav-menu__chevron" aria-hidden="true">&rarr;</span></a>';
		}
		public function end_el(&$output, $item, $depth = 0, $args = null) {
			if ($this->cartella($item, $depth)) { return; }
			$output .= '</li>';
		}
	}
}

/* -----------------------------------------------------------
 * 4) Menu di scorta (finche' non assegni un menu vero)
 *    Struttura ad albero: [titolo, url, [figli...]]
 * --------------------------------------------------------- */
if (!function_exists('tdtnav_albero')) {
	function tdtnav_albero() {
		return array(
			array('Homepage', home_url('/'), array()),
			array('Negozio', home_url('/negozio/'), array(
				array('Scopri i prodotti', home_url('/negozio/')),
				array('Carrello', home_url('/carrello/')),
				array('Pagamento', home_url('/pagamento/')),
			)),
			array('Contatti', home_url('/contatti/'), array()),
			array('Forniture B2B', home_url('/forniture-cuscini-b2b/'), array()),
		);
	}
}
if (!function_exists('tdtnav_scorta_desktop')) {
	function tdtnav_scorta_desktop() {
		$out = '<ul class="dtnav__menu">';
		foreach (tdtnav_albero() as $v) {
			$has = !empty($v[2]);
			$out .= '<li class="dtnav__item' . ($has ? ' dtnav__item--has' : '') . '">';
			$out .= '<a href="' . esc_url($v[1]) . '" class="dtnav__link">' . esc_html($v[0]);
			if ($has) { $out .= '<span class="dtnav__caret" aria-hidden="true">&#9662;</span>'; }
			$out .= '</a>';
			if ($has) {
				$out .= '<ul class="dtnav__sub">';
				foreach ($v[2] as $f) {
					$out .= '<li><a href="' . esc_url($f[1]) . '" class="dtnav__sublink">' . esc_html($f[0]) . '</a></li>';
				}
				$out .= '</ul>';
			}
			$out .= '</li>';
		}
		return $out . '</ul>';
	}
}
if (!function_exists('tdtnav_scorta_mobile')) {
	function tdtnav_scorta_mobile() {
		$out = '<ul class="dtnav-menu__list">';
		$i = 0;
		/* come nel walker: le voci con figli non si stampano, al loro
		   posto vanno i figli, tutti sullo stesso piano */
		foreach (tdtnav_albero() as $v) {
			$voci = !empty($v[2]) ? $v[2] : array(array($v[0], $v[1]));
			foreach ($voci as $f) {
				$i++;
				$indice = str_pad($i, 2, '0', STR_PAD_LEFT);
				$out .= '<li class="dtnav-menu__item"><a href="' . esc_url($f[1]) . '" class="dtnav-menu__link" data-dtnav-close>'
				      . '<span class="dtnav-menu__index">' . esc_html($indice) . '</span>'
				      . '<span class="dtnav-menu__label">' . esc_html($f[0]) . '</span>'
				      . '<span class="dtnav-menu__chevron" aria-hidden="true">&rarr;</span></a></li>';
			}
		}
		return $out . '</ul>';
	}
}

/* -----------------------------------------------------------
 * 5) L-ICONA DEL CARRELLO
 *
 *    Il numero lo stampa il server, cosi' e' gia' giusto al primo
 *    disegno. Poi resta aggiornato da solo in tre modi, dal piu'
 *    veloce al piu' sicuro:
 *      a) al clic su "aggiungi al carrello" il numero sale subito,
 *         senza aspettare la risposta (e' il "si accende immediato");
 *      b) WooCommerce rimpiazza il pallino con quello vero tramite i
 *         suoi frammenti (woocommerce_add_to_cart_fragments, sotto);
 *      c) se i frammenti non ci sono - o la pagina arriva dalla cache
 *         di SiteGround - il codice chiede il numero alla Store API.
 *    Senza WooCommerce l-icona non viene stampata affatto.
 * --------------------------------------------------------- */
if (!function_exists('tdtnav_cart_conta')) {
	function tdtnav_cart_conta() {
		if (!function_exists('WC')) { return 0; }
		$wc = WC();
		if (!is_object($wc) || empty($wc->cart) || !is_object($wc->cart)) { return 0; }
		if (!method_exists($wc->cart, 'get_cart_contents_count')) { return 0; }
		return (int) $wc->cart->get_cart_contents_count();
	}
}

/** Il pallino con il numero: e' questo il pezzo che WooCommerce rimpiazza. */
if (!function_exists('tdtnav_cart_pallino')) {
	function tdtnav_cart_pallino($n) {
		$n = (int) $n;
		return '<span class="dtnav__cart-n" data-count="' . esc_attr($n) . '" aria-hidden="true">'
		     . esc_html($n > 99 ? '99+' : (string) $n) . '</span>';
	}
}

/** L-icona intera (link alla pagina carrello + disegno + pallino). */
if (!function_exists('tdtnav_cart_icona')) {
	function tdtnav_cart_icona() {
		if (!function_exists('WC')) { return ''; }

		$n   = tdtnav_cart_conta();
		$url = function_exists('wc_get_cart_url') ? wc_get_cart_url() : home_url('/carrello/');
		$lab = ($n === 1) ? 'Carrello, 1 articolo' : sprintf('Carrello, %d articoli', $n);

		$svg = '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.4" '
		     . 'stroke-linecap="round" stroke-linejoin="round" aria-hidden="true" focusable="false">'
		     . '<path d="M3 4h2.2l2.3 9.6a1.8 1.8 0 0 0 1.75 1.4h7.9a1.8 1.8 0 0 0 1.75-1.35L21 7.4H6.4"/>'
		     . '<circle cx="10" cy="19.4" r="1.25"/><circle cx="17.3" cy="19.4" r="1.25"/></svg>';

		return '<a href="' . esc_url($url) . '" class="dtnav__cart' . ($n > 0 ? ' is-full' : '') . '"'
		     . ' aria-label="' . esc_attr($lab) . '" data-dtnav-cart>'
		     . $svg
		     . tdtnav_cart_pallino($n)
		     . '<span class="dtnav__cart-sr" role="status" aria-live="polite"></span>'
		     . '</a>';
	}
}

/** L-icona dell'account: porta a "Il mio account" (accesso o registrazione). */
if (!function_exists('tdtnav_account_icona')) {
	function tdtnav_account_icona() {
		if (!function_exists('wc_get_page_permalink')) { return ''; }

		$url = wc_get_page_permalink('myaccount');
		$lab = is_user_logged_in() ? 'Il tuo account' : 'Accedi o crea un account';

		$svg = '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.4" '
		     . 'stroke-linecap="round" stroke-linejoin="round" aria-hidden="true" focusable="false">'
		     . '<circle cx="12" cy="8.4" r="3.7"/><path d="M4.9 20c.9-3.7 3.8-5.8 7.1-5.8s6.2 2.1 7.1 5.8"/></svg>';

		return '<a href="' . esc_url($url) . '" class="dtnav__account" aria-label="' . esc_attr($lab) . '">'
		     . $svg . '</a>';
	}
}

/**
 * Il posto del selettore della lingua, subito prima dell'account.
 * Lo riempie il plugin TDT Lingue (cerca da solo [data-tdt-lingue]):
 * qui c-e' solo il contenitore. Dove il plugin non traduce (pagine di

	function tdtnav_lingua_posto() {
		if (!function_exists('tdt_lingue_attivo_qui') || !tdt_lingue_attivo_qui()) { return ''; }
		return '<div class="dtnav__lang" data-tdt-lingue></div>';
	}
}

/** Il pallino aggiornato viaggia insieme agli altri frammenti di Woo. */
if (!function_exists('tdtnav_cart_frammento')) {
	function tdtnav_cart_frammento($frammenti) {
		if (!is_array($frammenti)) { $frammenti = array(); }
		$frammenti['span.dtnav__cart-n'] = tdtnav_cart_pallino(tdtnav_cart_conta());
		return $frammenti;
	}
}
add_filter('woocommerce_add_to_cart_fragments', 'tdtnav_cart_frammento');

/** I frammenti hanno bisogno del loro script: su alcune pagine Woo non lo mette. */
if (!function_exists('tdtnav_accoda')) {
	function tdtnav_accoda() {
		if (is_admin() || !function_exists('WC')) { return; }
		if (wp_script_is('wc-cart-fragments', 'registered')) {
			wp_enqueue_script('wc-cart-fragments');
		}
	}
}
add_action('wp_enqueue_scripts', 'tdtnav_accoda', 20);

/* -----------------------------------------------------------
 * 6) Header + pannello del telefono
 *    Tutto costruito come stringa e stampato in un colpo solo: niente
 *    uscite e rientri dalla modalita' PHP in mezzo al codice, cosi' il
 *    frammento si comporta allo stesso modo dentro Code Snippets,
 *    WPCode o functions.php.
 * --------------------------------------------------------- */
if (!function_exists('tdtnav_header_html')) {
	function tdtnav_header_html() {
		$home = esc_url(home_url('/'));
		$cta  = esc_url(home_url('/negozio/'));   /* "Acquista" -> pagina Negozio */
		$cart = tdtnav_cart_icona();

		$desktop = '';
		$mobile  = '';
		if (has_nav_menu('dtnav') && class_exists('TDTNav_Desktop_Walker')) {
			$desktop = wp_nav_menu(array(
				'theme_location' => 'dtnav',
				'container'      => false,
				'items_wrap'     => '<ul class="dtnav__menu">%3$s</ul>',
				'walker'         => new TDTNav_Desktop_Walker(),
				'depth'          => 2,
				'echo'           => false,
			));
			$mobile = wp_nav_menu(array(
				'theme_location' => 'dtnav',
				'container'      => false,
				'items_wrap'     => '<ul class="dtnav-menu__list">%3$s</ul>',
				'walker'         => new TDTNav_Mobile_Walker(),
				'depth'          => 2,
				'echo'           => false,
			));
		}
		if (!is_string($desktop) || $desktop === '') { $desktop = tdtnav_scorta_desktop(); }
		if (!is_string($mobile)  || $mobile  === '') { $mobile  = tdtnav_scorta_mobile(); }

		$marchio = '<span class="dtnav__brand-the">The</span>'
		         . '<span class="dtnav__brand-name">Double Twenty</span>';

		$h  = '<header class="dtnav" id="dtnav">';
		$h .=   '<a href="' . $home . '" class="dtnav__brand" aria-label="The Double Twenty">' . $marchio . '</a>';
		$h .=   '<nav class="dtnav__links" aria-label="Navigazione principale">' . $desktop . '</nav>';
		$h .=   '<div class="dtnav__actions">';
		$h .=     tdtnav_lingua_posto();
		$h .=     tdtnav_account_icona();
		$h .=     $cart;
		$h .=     '<a href="' . $cta . '" class="dtnav__cta">Acquista</a>';
		$h .=     '<button type="button" class="dtnav__toggle" id="dtnav-toggle" '
		        . 'aria-label="Apri il menu" aria-expanded="false" aria-controls="dtnav-mobile">'
		        . '<span class="dtnav__toggle-line"></span><span class="dtnav__toggle-line"></span></button>';
		$h .=   '</div>';
		$h .= '</header>';

		$h .= '<div class="dtnav-menu" id="dtnav-mobile" role="dialog" aria-modal="true" aria-label="Menu" hidden>';
		$h .=   '<div class="dtnav-menu__bar">';
		$h .=     '<a href="' . $home . '" class="dtnav__brand" aria-label="The Double Twenty" data-dtnav-close>' . $marchio . '</a>';
		$h .=     '<button type="button" class="dtnav-menu__close" id="dtnav-close" aria-label="Chiudi il menu">'
		        . '<span class="dtnav-menu__close-x" aria-hidden="true"></span></button>';
		$h .=   '</div>';
		$h .=   $mobile;
		$h .=   '<div class="dtnav-menu__foot">';
		$h .=     '<a href="' . $cta . '" class="dtnav-menu__cta" data-dtnav-close>Acquista <span aria-hidden="true">&rarr;</span></a>';
		$h .=   '</div>';
		$h .= '</div>';

		return $h;
	}
}

if (!function_exists('tdtnav_js')) {
function tdtnav_js() {
	return <<<'JS'
(function(){
  /* ---------------- pannello del telefono ---------------- */
  var toggle=document.getElementById("dtnav-toggle");
  var closeBtn=document.getElementById("dtnav-close");
  var menu=document.getElementById("dtnav-mobile");
  if(toggle&&menu&&!menu.getAttribute("data-dt-init")){
    menu.setAttribute("data-dt-init","1");
    var openState=false;

    var openMenu=function(){
      if(openState)return; openState=true;
      menu.hidden=false; void menu.offsetWidth;
      menu.classList.add("is-open");
      document.documentElement.classList.add("dtnav-open");
      toggle.setAttribute("aria-expanded","true");
      if(closeBtn)closeBtn.focus();
    };
    var closeMenu=function(){
      if(!openState)return; openState=false;
      menu.classList.remove("is-open");
      document.documentElement.classList.remove("dtnav-open");
      toggle.setAttribute("aria-expanded","false");
      var done=function(){ if(!openState)menu.hidden=true; menu.removeEventListener("transitionend",done); };
      menu.addEventListener("transitionend",done);
      setTimeout(function(){ if(!openState)menu.hidden=true; },600);
      toggle.focus();
    };

    toggle.addEventListener("click",openMenu);
    if(closeBtn)closeBtn.addEventListener("click",closeMenu);
    var chiudi=menu.querySelectorAll("[data-dtnav-close]");
    for(var i=0;i<chiudi.length;i++){ chiudi[i].addEventListener("click",closeMenu); }

    /* accordion: tocco su una voce con figli apre/chiude il sottomenu */
    var tasti=menu.querySelectorAll(".dtnav-menu__toggle");
    for(var k=0;k<tasti.length;k++){
      (function(btn){
        btn.addEventListener("click",function(){
          var li=btn.closest?btn.closest(".dtnav-menu__item--has"):btn.parentNode;
          if(!li)return;
          var aperto=li.classList.toggle("is-expanded");
          btn.setAttribute("aria-expanded",aperto?"true":"false");
        });
      })(tasti[k]);
    }

    document.addEventListener("keydown",function(e){ if(e.key==="Escape"&&openState)closeMenu(); });
  }

  /* ---------------- barra bianca quando si scorre ----------------
     Il segno "data-tdt-barra" e' lo stesso che usava il vecchio frammento
     tdt-menu-barra: se per un po' restano attivi tutti e due, il secondo
     vede il segno e lascia fare al primo. */
  var barra=document.querySelector("header.dtnav")||document.querySelector(".dtnav");
  if(barra&&!barra.getAttribute("data-tdt-barra")){
    barra.setAttribute("data-tdt-barra","1");
    var acceso=null,inCoda=false;
    var guarda=function(){
      inCoda=false;
      var y=window.pageYOffset||document.documentElement.scrollTop||0;
      var deve=y>8;                    /* dopo 8 px di scorrimento */
      /* si tocca la pagina solo quando lo stato cambia davvero: durante
         uno scorrimento lungo sarebbero centinaia di scritture inutili */
      if(deve===acceso)return;
      acceso=deve;
      if(deve)barra.classList.add("dtnav--barra");
      else barra.classList.remove("dtnav--barra");
    };
    /* il conto si fa una volta per fotogramma, non a ogni scatto della
       rotellina: cosi' lo scorrimento resta liscio */
    var chiedi=function(){
      if(inCoda)return;
      inCoda=true;
      window.requestAnimationFrame(guarda);
    };
    window.addEventListener("scroll",chiedi,{passive:true});
    window.addEventListener("resize",chiedi,{passive:true});
    window.addEventListener("load",guarda);
    guarda();                          /* pagina riaperta gia' scorsa */
  }

  /* ---------------- icona del carrello ----------------
     Il numero si accende appena si clicca "aggiungi al carrello" e poi
     viene confermato dal server. Vedi il commento al punto 5 del PHP. */
  var CFG=window.DTNAV_CART||{};
  var cart=document.querySelector(".dtnav__cart");
  if(!cart||cart.getAttribute("data-dt-cart"))return;
  cart.setAttribute("data-dt-cart","1");

  function pallino(){ return cart.querySelector(".dtnav__cart-n"); }
  function leggi(){
    var b=pallino();
    if(!b)return 0;
    var n=parseInt(b.getAttribute("data-count"),10);
    return isNaN(n)?0:n;
  }

  var nodo=pallino();      /* serve a capire se Woo ha rimpiazzato il pallino */
  var ultimo=leggi();
  var attesa=null;
  var risposto=false;

  function lampo(){
    cart.classList.remove("is-pop");
    void cart.offsetWidth;            /* riavvia l-animazione anche a meta- corsa */
    cart.classList.add("is-pop");
    clearTimeout(cart._pop);
    cart._pop=setTimeout(function(){ cart.classList.remove("is-pop"); },900);
  }

  function scrivi(n,accendi){
    n=n<0?0:n;
    var b=pallino();
    if(b){
      b.setAttribute("data-count",n);
      b.textContent=n>99?"99+":String(n);   /* testo, mai HTML */
      nodo=b;
    }
    if(n>0)cart.classList.add("is-full"); else cart.classList.remove("is-full");
    cart.setAttribute("aria-label",n===1?"Carrello, 1 articolo":"Carrello, "+n+" articoli");
    var sr=cart.querySelector(".dtnav__cart-sr");
    if(sr)sr.textContent=n===0?"":(n===1?"1 articolo nel carrello":n+" articoli nel carrello");
    ultimo=n;
    if(accendi)lampo();
  }

  /* Il numero vero, chiesto al negozio: rete di sicurezza quando i
     frammenti di WooCommerce non ci sono o la pagina viene dalla cache. */
  function dalServer(){
    if(!window.fetch||!CFG.cart)return;
    window.fetch(CFG.cart,{credentials:"same-origin",headers:{"Accept":"application/json"}})
      .then(function(r){ return r.ok?r.json():null; })
      .then(function(j){
        if(!j||typeof j.items_count!=="number")return;
        risposto=true;
        if(j.items_count!==ultimo)scrivi(j.items_count,j.items_count>ultimo);
      })["catch"](function(){});
  }

  /* a) il numero sale subito, senza aspettare la risposta */
  document.addEventListener("click",function(e){
    var b=e.target&&e.target.closest?
      e.target.closest(".add_to_cart_button,.ajax_add_to_cart,.single_add_to_cart_button"):null;
    if(!b||!b.classList)return;
    if(b.classList.contains("disabled")||b.classList.contains("wc-variation-selection-needed"))return;
    var q=1;
    var form=b.closest?b.closest("form.cart"):null;
    var input=form?form.querySelector("input.qty"):null;
    if(input&&input.value)q=parseInt(input.value,10)||1;
    else if(b.getAttribute("data-quantity"))q=parseInt(b.getAttribute("data-quantity"),10)||1;
    if(!q||q<1)q=1;
    scrivi(ultimo+q,true);
    clearTimeout(attesa);
    attesa=setTimeout(dalServer,2200);
  },true);

  /* b) il numero del server arriva con i frammenti di WooCommerce.
        Il controllo si fa al giro dopo, perche' il pallino viene
        rimpiazzato da un altro ascoltatore dello stesso evento.
        "updated_wc_div" lo manda la pagina carrello nostra (tdt-carrello):
        li' i frammenti non passano, quindi si chiede il numero al negozio. */
  if(window.jQuery){
    window.jQuery(document.body).on(
      "added_to_cart removed_from_cart wc_fragments_refreshed wc_fragments_loaded updated_cart_totals updated_wc_div",
      function(){
        setTimeout(function(){
          var b=pallino();
          if(b&&b!==nodo){             /* il frammento ha portato il numero vero */
            nodo=b;
            risposto=true;
            clearTimeout(attesa);
            var n=leggi();
            if(n!==ultimo)scrivi(n,n>ultimo); else ultimo=n;
            return;
          }
          clearTimeout(attesa);        /* nessun frammento: lo chiediamo noi */
          attesa=setTimeout(dalServer,150);
        },0);
      }
    );
  }

  /* c) pagina servita dalla cache con un carrello gia' pieno */
  setTimeout(function(){
    if(risposto||ultimo>0)return;
    if(!/wp_woocommerce_session_|woocommerce_items_in_cart/.test(document.cookie))return;
    dalServer();
  },1500);
  window.addEventListener("pageshow",function(e){ if(e.persisted)dalServer(); });
})();
JS;
}
